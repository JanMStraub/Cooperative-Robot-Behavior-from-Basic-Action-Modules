\section{The Problem}

\begin{frame}{The Reasoning--Execution Gap}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\footnotesize
			\begin{itemize}
				\item Classical robots: precise, but caged. Rigid paths, no tolerance for changes.
				\item VLMs bring world knowledge, but cannot reliably turn reasoning into \textit{physical} action.
				\item Models often ignore past physical state when planning the next move.
			\end{itemize}
			\begin{block}{Why cooperation is the hard case}
				Two arms must \textbf{avoid collision}, \textbf{share a workspace}, and \textbf{synchronize timing}.
			\end{block}
		\end{column}
		\begin{column}{0.4\textwidth}
			\centering
			\includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{14_robot_env.png}\\
			\footnotesize Two AR4~\cite{AnninRobotics2026} arms, a shared workspace, and a stereo camera in Unity~\cite{unity}.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Background}
	\begin{itemize}
		\item \textbf{End-to-end VLA} (RT-2)~\cite{brohan2023rt2visionlanguageactionmodelstransfer}: great generalization, but data-hungry and opaque.
		\item \textbf{Code-as-Policies}~\cite{liang2023codepolicieslanguagemodel}: compositional and transparent, but tied to its API set.
		\item \textbf{Framework-scale} (AutoRT)~\cite{ahn2024autortembodiedfoundationmodels}: practical scaffolding, but no joint reasoning over a \textit{shared} object.
		\item \textbf{RoCo}~\cite{mandi2023rocodialecticmultirobotcollaboration}: per-robot agents negotiate, but free-form way-points.
	\end{itemize}
	\vfill
	\begin{block}{Research gap}
		\textit{Two independent} arms compose object coordination from a fixed \textit{grounded} vocabulary of operations, on a local compact model.
	\end{block}
\end{frame}

\begin{frame}{Two Design Commitments}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\begin{block}{Run the model \textit{locally}}
				\begin{itemize}\small
					\item No cloud cost, no external dependency.
					\item No sensitive scene data leaving the machine.
					\item A compact VLM is enough.
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{block}{Ground the model in \textit{modular operations}}
				\begin{itemize}\small
					\item Reasoning restricted to a vocabulary of \textit{executable} actions.
					\item Separates semantic understanding from physical execution.
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\centering\small Default backend model: \textbf{Magistral Small 2509 (24B, 4-bit)~\cite{mistral_magistral_small_2025}}
\end{frame}

\section{Pillar 1: Grounding}

\begin{frame}{System Architecture}
	\centering
	\includegraphics[height=0.72\textheight]{16_system_flow.png}\\
	\scriptsize Built on Unity, ROS~2~\cite{Macenski_2022}, and MoveIt~\cite{coleman2014reducingbarrierentrycomplex}.
\end{frame}

\begin{frame}{The Operation Pool}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\footnotesize
			Operations organized by \textbf{complexity}, four tiers:
			\begin{description}\footnotesize
				\item[Atomic (8)] indivisible, no perception: gripper, signal, wait.
				\item[Basic (6)] single dispatch, introduces perception.
				\item[Intermediate (4)] composite single-robot: placement, stereo detection, scene analysis.
				\item[Complex (2)] full pipelines: grasp planning, handoff reception.
			\end{description}
			\vspace{0.3em}
			\textbf{Why it grounds:} model can only express plans the robot can run.
			\\[0.3em]
			{\scriptsize\color{black!70} \textbf{RAG Index (31 items):} Alongside the 20 operations, RAG indexes 7 workflow patterns and 4 context documents to supply multi-step exemplars and spatial priors.}
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{09_operation_map.png}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Reflection Recovery}
	\begin{block}{Gated retry on failure}
		\begin{itemize}
			\item On an eligible failure, re-query the model with a structured \textit{hint} (Reflexion-style verbal feedback~\cite{shinn2023reflexionlanguageagentsverbal}).
			\item Eligibility is gated to Navigation / Manipulation / Perception: no point re-querying a sync barrier.
		\end{itemize}
	\end{block}
	\vfill
	\footnotesize Optional knowledge-graph layer for spatial queries; ablation cost covered later.
\end{frame}

\begin{frame}{Grounding $\Rightarrow$ Reliability}
	\begin{columns}[c]
		\begin{column}{0.45\textwidth}
			\begin{itemize}\small
				\item \textbf{B1--B5}: \textbf{success over all 25 runs}, zero hallucinated ops, zero retries.
				\item \textit{Identical} operation sequence on all 5 runs of each benchmark.
			\end{itemize}
		\end{column}
		\begin{column}{0.55\textwidth}
			\centering
			\includegraphics[width=\linewidth]{21_success_rate_by_model_b1-b5.png}\\
			\footnotesize Magistral, Ministral, and both Gemma~4 models clear the full single-robot gradient; the Qwen3-VL models fail from B3 onward by truncating plans.
		\end{column}
	\end{columns}
\end{frame}

\section{Pillar 2: Cooperation}

\begin{frame}{The Model Composes}
	\begin{block}{Two general primitives}
		\begin{itemize}
			\item \texttt{signal} / \texttt{wait\_for\_signal}: one robot pauses at a checkpoint until the other signals readiness.
			\item \textbf{Parallel groups}: operations dispatched concurrently when no ordering is needed.
		\end{itemize}
	\end{block}
	\vfill
	The executor has \textbf{no} coordination policy, but physical coordination (approach geometry, release timing) is handled in the operations, not the plan.
\end{frame}

\begin{frame}{The Handoff}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Default model: 5/5 runs; all operations first try.
				\item Ordering and the signal/wait barrier are set by a prompt rule.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{18_handoff_sequence_example.png}\\
			\footnotesize Handoff sequence.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Negotiation}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\footnotesize
			A single model planning both arms under-specifies roles; \textit{implicit conflicts} surface only at execution.
			\vfill
			\textbf{Design choices}:
			\begin{itemize}\setlength{\itemsep}{2pt}
				\item Written accept/reject with justifications, so disagreement is resolved in language, before any motion.
				\item Proposal rotates round-robin; \textbf{unanimity}. A plan omitting any robot is rejected outright.
				\item The silent fallback keeps negotiation an \textit{option}, not a dependency.
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\centering
			\includegraphics[height=0.8\textheight]{19_negotiation.png}
		\end{column}
	\end{columns}
\end{frame}

	\centering
	\begin{table}\footnotesize
		\begin{tabular}{lcc}
			\toprule
			                & \textbf{Task SR}       & \textbf{Ops/run} \\
			\midrule
			Negotiation on  & \textbf{0.713} (57/80) & 35.8             \\

		\item Pays off when the single-shot plan is already close to correct; doesn't rescue weaker planners from role conflicts.
	\end{itemize}
\end{frame}

\section{Boundary}

\begin{frame}{Concurrent Bimanual Lift}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{All six models fail.} Three failure modes:
			\begin{enumerate}\footnotesize
				\item Signal primitives never dispatched once control crosses robots outside a parallel group.
				\item Final lift: two \textit{independently} chosen targets land too close together, violating the separation check.
				\item Deeper sim limit: a Unity \texttt{Transform} has one parent, so a single object can't attach to two grippers at once.
			\end{enumerate}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{block}{Gap}
				\footnotesize
				\textbf{Partner-aware target selection}. Each arm picks a valid target; nothing checks one \textit{against the other}.
			\end{block}
			\centering
			\includegraphics[width=0.8\linewidth]{22_bilift_sketch.png}\\
			\scriptsize Sketch of the envisioned concurrent bimanual lift.
		\end{column}
	\end{columns}
\end{frame}

\section{Results}
\begin{frame}{Ablation Synthesis}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\footnotesize
			\begin{itemize}
				\item \textbf{ROS/MoveIt +32 pp}: Unity-only grasp timeouts; plus collision-aware planning.
				\item \textbf{Reflection +14 pp}: clears boundary-reach failures; doubles time when it fires.
				\item \textbf{Negotiation +10 pp}: default model only; neutral/negative on the other five.
				\item \textbf{VGN~\cite{breyer2021volumetricgraspingnetworkrealtime} +14 pp} (+35 pp on grasp-held): center-of-object grasp, not predicted orientation.
				\item \textbf{RAG / KG: net cost}: prompt over-anchoring, helps no backend.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{08_ablation.png}\\
			\scriptsize Per-component deltas (default model; $n=20$ for B11--B13, B16; $n=5$ for B14--B15).
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Limitations}
	\begin{columns}[t]
		\begin{column}{0.33\textwidth}
			\begin{block}{Sensing}
				\footnotesize Cube features too small to match, so the \textit{dense} stereo cloud is sparse; per-object localization stays robust.
			\end{block}
		\end{column}
		\begin{column}{0.33\textwidth}
			\begin{block}{Latency}
				\footnotesize VLM parse $\approx$ \textbf{20\,s} median prevents \textit{reactive} control; re-plan between motions, never during.
			\end{block}
		\end{column}
		\begin{column}{0.33\textwidth}
			\begin{block}{Sim-to-real}
				\footnotesize Robot joints and grasp-force thresholds tuned for sim physics, so they need revalidation on hardware.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Five Contributions}
	\begin{enumerate}
		\item \textbf{Grounded Control Framework}: A hierarchical 20-operation registry, retrieved on demand, turning language into verified behavior with no fine-tuning.
		\item \textbf{Runtime-Composed Coordination}: The model \textit{sequences} handoff steps and parallelizes independent tasks with \texttt{signal}/\texttt{wait} and parallel groups.
		\item \textbf{Capability-Gated Negotiation}: Helps only already capable planners, while degrading weaker models. (\textit{adapting RoCo~\cite{mandi2023rocodialecticmultirobotcollaboration}})
		\item \textbf{Operation-Gated Reflection}: Uses semantic repair to fix physical execution errors. (\textit{adapting Reflexion~\cite{shinn2023reflexionlanguageagentsverbal}})
		\item \textbf{Two-Layer Safety Adaptation}: Adapts AutoRT~\cite{ahn2024autortembodiedfoundationmodels} to fixed-base manipulators by pairing an LLM semantic constitution with a hard kinematic gate the model cannot override.
	\end{enumerate}
\end{frame}

\begin{frame}{Future Work}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{Deployment}
			\begin{itemize}\footnotesize
				\item \texttt{--env real} swaps camera adapter at the abstraction boundary.
				\item Control switches to ROS/MoveIt collision-aware planning, already configured for dual-robot operation.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Remaining work}
			\begin{itemize}\footnotesize
				\item \textbf{Depth}: LIDAR or per-robot stereo unlocks VGN-grade grasping.
				\item \textbf{Bimanual}: partner-aware target selection closes the boundary gap.
				\item \textbf{Faster inference}: distilled task-specific models or speculative decoding to shrink the 20\,s window.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Demo \& Questions}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\footnotesize
			\textbf{What works}
			\begin{itemize}\scriptsize
				\item Single-robot: 25/25 runs.
				\item Long chain: 85 ops/run, 5 cycles, no cascading errors.
				\item Handoff: works; sync barrier correctly realized.
				\item Parallelism: works for 4/6 backends.
				\item Safety gate: no false-accept, reproducible.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}

			\end{itemize}

			\begin{itemize}\scriptsize

			\end{itemize}
		\end{column}

		\begin{column}{0.46\textwidth}
			\footnotesize\textbf{General rule in the prompt}
			\begin{block}{}\scriptsize\ttfamily
				Same parallel\_group = concurrent.\\
				Later group waits for all prior.\\
				VAR DEPENDENCY LAW: if B reads a\\
				\$var captured by A, B must be in a\\
				strictly higher parallel\_group.
			\end{block}
		\end{column}
		\begin{column}{0.52\textwidth}
			\footnotesize\textbf{B10 plan the model emitted}
			\begin{block}{}\scriptsize\ttfamily
				g1: detect(R1,blue) $\parallel$ detect(R2,green)\\
				g2: grasp(R1) $\parallel$ grasp(R2)\\
				g3: field(R1,A) $\parallel$ field(R2,B)\\

\begin{frame}[noframenumbering]{Backup: What VGN changes}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{table}\scriptsize
				\begin{tabular}{lcc}
					\toprule
					\textbf{Held \& lifted} & \textbf{VGN on}        & \textbf{VGN off} \\
					\midrule
					Magistral               & \textbf{14/20 (0.700)} & 7/20 (0.350)     \\
					Pooled, 6 models        & \textbf{0.750}         & 0.417            \\
					\bottomrule

			\footnotesize
			\begin{itemize}\footnotesize
				\item Every backend improves: +20\,pp (Ministral) to +65\,pp (Gemma-4B).
				\item Success = held and lifted clear of the table.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{block}{Mechanism: position, not orientation}
				\footnotesize
				\begin{itemize}\footnotesize
					\item \textbf{Same orientation both conditions}: top-down, yaw from the object's long axis. VGN's own poses fail motion planning and are discarded.
					\item \textbf{The difference is position}: VGN grasps the detected object \textit{center}; the baseline's contact point sits off-center on angled cubes and slips.
				\end{itemize}
			\end{block}
			\begin{block}{Scope}
				\footnotesize Position is the stereo-written WorldState estimate, so B15 isolates grasp \textit{positioning}. Recovering VGN's learned pose needs a real depth sensor.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Collision Safety Layers}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\footnotesize
			\begin{enumerate}\footnotesize
				\item \textbf{Static pre-check} (Python, before dispatch): checks workspace bounds, object conflicts, and deadlocks on the planned targets.
				\item \textbf{Workspace regions}: left / right / shared zones; a contested zone is handed over explicitly via \texttt{yield\_workspace}.
				\item \textbf{Runtime guard} (Unity, every physics step): \texttt{ProximityGuard} freezes motion at 0.25\,m end-effector distance, resumes at 0.35\,m; link-to-link limit 0.15\,m.
			\end{enumerate}
		\end{column}
		\begin{column}{0.45\textwidth}
			\centering
			\includegraphics[width=\linewidth]{15_robot_workspaces.png}
		\end{column}
	\end{columns}
	\vspace{0.4em}
	The LLM is never trusted for safety: layers 1 and 3 are code, not reasoning.
\end{frame}
