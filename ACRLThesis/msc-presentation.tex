\begin{frame}{Condensed Results}
	\begin{block}{Thesis in one sentence}
		A \textbf{locally hosted vision-language model (VLM)}, grounded in a structured pool of \textbf{20 basic operations}, turns natural language into reliable \textbf{robot behavior}, with \textbf{no fine-tuning}.
	\end{block}
	\vfill
	\begin{columns}[t]
		\begin{column}{0.33\textwidth}

\section{The Problem}

\begin{frame}{The Reasoning--Execution Gap}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\footnotesize
			\begin{itemize}
				\item Classical robots: precise, but caged. Rigid paths, no tolerance for changes.
				\item VLMs bring world knowledge and reasoning, but cannot reliably turn abstract reasoning into \emph{physical} action.
				\item Models often ignore past physical state when planning the next move.
			\end{itemize}
			\begin{block}{Why cooperation is the hard case}
				Two arms must \textbf{avoid collision}, \textbf{share a workspace}, and \textbf{synchronize timing}.
			\end{block}
		\end{column}
		\begin{column}{0.4\textwidth}
			\centering
			\includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{14_robot_env.png}\\
			\footnotesize Two AR4~\cite{AnninRobotics2026} arms, a shared workspace, and a stereo camera in Unity~\cite{unity}.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Two Design Commitments}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\begin{block}{Run the model \emph{locally}}
				\begin{itemize}\small
					\item No cloud cost, no external dependency.
					\item No sensitive scene data leaving the machine.
					\item A compact VLM is enough; no frontier model required.
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{block}{Ground the model in \emph{modular operations}}
				\begin{itemize}\small
					\item Reasoning restricted to a vocabulary of \emph{executable} actions.
					\item Separates semantic understanding from physical execution, making it reliable.
					\item Vision resolves targets live; no task-specific training.
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}
	\vfill
	\centering\small Default backend model: \textbf{Magistral Small 2509 (24B, 4-bit)~\cite{mistral_magistral_small_2025}}
\end{frame}

\begin{frame}{Background}
	\begin{itemize}
		\item \textbf{End-to-end VLA} (RT-2, PaLM-E)~\cite{brohan2023rt2visionlanguageactionmodelstransfer,driess2023palmeembodiedmultimodallanguage}: great generalization, but data-hungry and opaque.
		\item \textbf{Code-as-Policies}~\cite{liang2023codepolicieslanguagemodel}: compositional and transparent, but tied to its API set.
		\item \textbf{Framework-scale} (AutoRT)~\cite{ahn2024autortembodiedfoundationmodels}: practical scaffolding, but no joint reasoning over a \emph{shared} object.
		\item \textbf{RoCo}~\cite{mandi2023rocodialecticmultirobotcollaboration}: per-robot agents negotiate, but free-form way-points.
	\end{itemize}
	\vfill
	\begin{block}{Research gap}
		Runtime composition of shared-object coordination from a fixed vocabulary of \emph{operations}, cooperation \emph{grounded} in reasoning rather than a fixed execution path, is comparatively unexplored.
	\end{block}
\end{frame}

\section{Pillar 1: Grounding}

\begin{frame}{System Architecture}
	\centering
	\includegraphics[height=0.75\textheight]{16_system_flow.png}\\
	\scriptsize Built on Unity, ROS~2~\cite{Macenski_2022}, and MoveIt~\cite{coleman2014reducingbarrierentrycomplex}.
\end{frame}

\begin{frame}{The Operation Pool}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\footnotesize
			Operations organized by \textbf{complexity}, four tiers:
			\begin{description}\footnotesize
				\item[Atomic (8)] indivisible, no perception: gripper, signal, wait.
				\item[Basic (6)] single dispatch, introduces perception.
				\item[Intermediate (4)] composite single-robot: placement, stereo detection, scene analysis.
				\item[Complex (2)] full pipelines: grasp planning, handoff reception.
			\end{description}
			\vspace{0.3em}
			\textbf{Why it grounds:} model can only express plans the robot can actually run.
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{09_operation_map.png}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Reflection Recovery}
	\begin{block}{Gated retry on failure}
		\begin{itemize}
			\item On an eligible failure, re-query the model with a structured \emph{hint} (Reflexion-style verbal feedback~\cite{shinn2023reflexionlanguageagentsverbal}), closing the perception--action loop at the language level.
			\item Eligibility is gated to Navigation / Manipulation / Perception: no point re-querying a sync barrier.
		\end{itemize}
	\end{block}
	\vfill
	\footnotesize An optional knowledge-graph layer (spatial reachability / handoff queries) is also available; its ablation cost is covered later.
\end{frame}

\begin{frame}{Grounding $\Rightarrow$ Reliability}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}\small
				\item \textbf{B1--B5} (navigate $\to$ grasp $\to$ full pick-and-place): \textbf{success over all 25 runs}, zero hallucinated ops, zero retries.
				\item \emph{Identical} operation sequence on all 5 runs of each benchmark.
				\item \textbf{B8} (long chain): 20 sub-tasks, \textbf{85 operations}/run, 5 cycles, all executed, no cascading errors.
			\end{itemize}
			\vspace{0.4em}
			\footnotesize Cost is dominated by motion primitives (\texttt{grasp\_object}, \texttt{place\_object}).
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{01_success_rate_by_model.png}\\
			\footnotesize Magistral, Ministral, and both Gemma~4 models clear the full single-robot gradient; the Qwen3-VL models fail from B3 onward by truncating plans.
		\end{column}
	\end{columns}
\end{frame}

\section{Pillar 2: Cooperation}

\begin{frame}{The Model Composes}
	\begin{block}{Two general primitives}
		\begin{itemize}
			\item \texttt{signal} / \texttt{wait\_for\_signal}: one robot pauses at a checkpoint until the other signals readiness.
			\item \textbf{Parallel groups}: operations dispatched concurrently when no ordering is needed.
		\end{itemize}
	\end{block}
	\vfill
	The executor holds \textbf{no} coordination policy. Guided by prompt rules and retrieved examples, the model places the synchronization barriers and parallel groups; the physical coordination, approach geometry and release timing, lives in the operations, not the plan.
\end{frame}

\begin{frame}{The Handoff}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Magistral and Ministral: 5/5 runs; all operations first try.
				\item Ordering and the signal/wait barrier are set by an prompt rule.
				\item Gemma-4B partial (3/5); Qwen3 and Gemma-2B fail at plan construction.
				\item \textbf{Parallelism}: the same primitives schedule independent pick-and-places (4/6 backends).
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{18_handoff_sequence_example.png}\\
			\footnotesize Handoff sequence.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Negotiation}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\footnotesize
			A single model planning both arms under-specifies roles; \emph{implicit conflicts} surface only at execution.
			\vfill
			\textbf{Protocol}:
			\begin{enumerate}\setlength{\itemsep}{0pt}
				\item \textbf{Analyze}: each agent assesses its own role (parallel).
				\item \textbf{Propose}: one agent/round drafts a \emph{complete} plan (round-robin).
				\item \textbf{Evaluate}: peers accept/reject with justification; \textbf{unanimity} required. Rejections feed the next proposal.
			\end{enumerate}
			No consensus in 3 rounds $\to$ single-model parsing.
		\end{column}
		\begin{column}{0.48\textwidth}
			\centering
			\includegraphics[height=0.8\textheight]{19_negotiation.png}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Negotiation Pays Off}
	\centering
	\begin{table}\footnotesize
		\begin{tabular}{lcc}
			\toprule
			                & \textbf{Task SR}       & \textbf{Ops/run} \\
			\midrule
			Negotiation on  & \textbf{0.713} (57/80) & 35.8             \\
			Negotiation off & 0.613 (49/80)          & 29.9             \\
			\bottomrule
		\end{tabular}
	\end{table}
	\vspace{0.6em}
	\begin{itemize}\footnotesize
		\item \textbf{+10 pp} task success for the \textbf{default model}.
		\item Roughly neutral for three backends; \emph{negative} for two ($-19$, $-21$ pp).
		\item Pays off when the single-shot plan is already close to correct; doesn't rescue weaker planners from role conflicts.
	\end{itemize}
\end{frame}

\section{Pillar 3: The Boundary}

\begin{frame}{Concurrent Bimanual Lift}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{All six models fail.} Three failure modes:
			\begin{enumerate}\footnotesize
				\item Signal primitives never dispatched once control crosses robots outside a parallel group.
				\item Final lift: two \emph{independently} chosen targets land too close together, violating the separation check.
				\item Deeper sim limit: a Unity \texttt{Transform} has one parent, so a single object can't attach to two grippers at once.
			\end{enumerate}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{block}{Gap}
				\footnotesize
				\textbf{Partner-aware target selection}. Each arm picks a valid target; nothing checks one \emph{against the other}.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Ablation Synthesis}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\footnotesize
			\begin{itemize}\itemsep2pt
				\item \textbf{ROS/MoveIt +32 pp}: Unity-only grasp timeouts; plus collision-aware planning.
				\item \textbf{Reflection +14 pp}: clears boundary-reach failures; doubles time when it fires.
				\item \textbf{Negotiation +10 pp}: default model only; neutral/negative on the other five.
				\item \textbf{VGN +14 pp} (+35 on grasp-held): center-of-object grasp, not predicted orientation.
				\item \textbf{RAG / KG: net cost}: prompt over-anchoring, helps no backend.
			\end{itemize}
		\end{column}
        \begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=\linewidth]{08_ablation.png}\\
			\scriptsize Per-component deltas (default model; $n=20$ for B11--B13, B16; $n=5$ for B14--B15).
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Limitations}
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			\begin{block}{Sensing}
				\footnotesize Cube features too small to match, so the stereo cloud is sparse.
			\end{block}
			\begin{block}{Latency}
				\footnotesize VLM parse $\approx$ \textbf{20\,s} median, which prevents \emph{reactive} control; re-plan only between motions, never during.
			\end{block}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{block}{Sim-to-real}
				\footnotesize Robot joint values all tuned for simulation, so they need revalidation on hardware.
			\end{block}
			\begin{block}{Validation}
				\footnotesize 5/6 models forward impossible inputs; the guard must be pre-parse.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Five Contributions}
	\begin{enumerate}\itemsep4pt
		\item \textbf{Grounded control framework}: a hierarchical 20-operation registry, retrieved on demand, turning language into verified behavior with no fine-tuning.
		\item \textbf{Runtime-composed coordination}: the model \emph{sequences} handoff steps and parallelizes independent tasks with \texttt{signal}/\texttt{wait} and parallel groups; the physical transfer itself is an engineered operation it invokes.
		\item \textbf{Consensus-gated negotiation}: helps only an already-capable planner (+10~pp default; neutral-to-negative for the rest).
		\item \textbf{Operation-gated reflection}: gated verbal repair; +14~pp for the default model.
		\item \textbf{AutoRT safety adaptation}: a hard kinematic gate the semantic check can't override.
	\end{enumerate}
\end{frame}

\begin{frame}{Future Work}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{Deployment}
			\begin{itemize}\footnotesize
				\item \texttt{--env real} swaps camera and hardware adapters at the abstraction boundary.
				\item Control switches to ROS/MoveIt collision-aware planning, already dual-robot configured.
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\textbf{Remaining work}
			\begin{itemize}\footnotesize
				\item \textbf{Depth}: LIDAR or per-robot stereo unlocks VGN-grade grasping; a SLAM volumetric map adds object permanence.
				\item \textbf{Force-torque}: a wrist sensor replaces the contact-duration heuristic.
				\item \textbf{Bimanual}: partner-aware target selection closes the boundary gap.
				\item \textbf{Pre-parse validation}: robot-ID / object-existence checks before any model call.
				\item \textbf{Faster inference}: distilled task-specific models or speculative decoding to shrink the 20\,s window.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

			\begin{block}{}\scriptsize\ttfamily
			Same parallel\_group = concurrent.\\
			Later group waits for all prior.\\
			VAR DEPENDENCY LAW: if B reads a\\
			\$var captured by A, B must be in a\\
			strictly higher parallel\_group.
			\end{block}
		\end{column}
		\begin{column}{0.52\textwidth}
			\footnotesize\textbf{B10 plan the model emitted}
			\begin{block}{}\scriptsize\ttfamily
			g1: detect(R1,blue) $\parallel$ detect(R2,green)\\
			g2: grasp(R1) $\parallel$ grasp(R2)\\
			g3: field(R1,A) $\parallel$ field(R2,B)\\

		\end{column}

	\end{itemize}
	\begin{exampleblock}{Takeaway}
		General rule in $\to$ task-specific parallel schedule out, with no matching example. The handoff, by contrast, follows an explicit ordering rule in the prompt.
	\end{exampleblock}
\end{frame}

\begin{frame}[noframenumbering]{Backup: What VGN~\cite{breyer2021volumetricgraspingnetworkrealtime} changes}
	\begin{columns}[T]

			\begin{table}\scriptsize

				\item Every backend improves: +20\,pp (Ministral) to +65\,pp (Gemma-4B).

				\footnotesize Position comes from the WorldState oracle, so B15 isolates grasp \emph{positioning}. Recovering VGN's learned pose needs a real depth sensor.
			\end{block}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Collision Safety Layers}
	\begin{columns}[T]
		\begin{column}{0.52\textwidth}
			\footnotesize
			\begin{enumerate}\footnotesize
				\item \textbf{Static pre-check} (Python, before dispatch): \texttt{CoordinationVerifier} checks workspace bounds, object conflicts, deadlock detection on the planned targets.
				\item \textbf{Workspace regions}: left / right / shared zones; a contested zone is handed over explicitly via \texttt{yield\_workspace}.
				\item \textbf{Runtime guard} (Unity, every physics step): \texttt{ProximityGuard} freezes motion at 0.25\,m end-effector distance, resumes at 0.35\,m; link-to-link limit 0.15\,m.
			\end{enumerate}
			\vspace{0.4em}
			The LLM is never trusted for safety: layers 1 and 3 are code, not reasoning.
		\end{column}
